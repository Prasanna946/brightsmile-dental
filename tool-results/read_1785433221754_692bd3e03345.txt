   459→
   460→interface FormData {
   461→  name: string;
   462→  phone: string;
   463→  email: string;
   464→  service: string;
   465→  preferredDate: string;
   466→  message: string;
   467→}
   468→
   469→function AppointmentSection({
   470→  sectionRef,
   471→}: {
   472→  sectionRef: React.RefObject<HTMLDivElement | null>;
   473→}) {
   474→  const { toast } = useToast();
   475→  const [form, setForm] = useState<FormData>({
   476→    name: "",
   477→    phone: "",
   478→    email: "",
   479→    service: "",
   480→    preferredDate: "",
   481→    message: "",
   482→  });
   483→  const [errors, setErrors] = useState<Partial<FormData>>({});
   484→  const [submitting, setSubmitting] = useState(false);
   485→  const [submitted, setSubmitted] = useState(false);
   486→
   487→  const validate = (): boolean => {
   488→    const e: Partial<FormData> = {};
   489→    if (!form.name.trim()) e.name = "Please enter your name.";
   490→    if (!form.phone.trim()) e.phone = "Please enter your phone number.";
   491→    if (!form.email.trim()) {
   492→      e.email = "Please enter your email.";
   493→    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
   494→      e.email = "Please enter a valid email.";
   495→    }
   496→    if (!form.service) e.service = "Please select a service.";
   497→    setErrors(e);
   498→    return Object.keys(e).length === 0;
   499→  };
   500→
   501→  const handleSubmit = async (e: FormEvent) => {
   502→    e.preventDefault();
   503→    if (!validate()) return;
   504→
   505→    setSubmitting(true);
   506→
   507→    try {
   508→      /* ---------------------------------------------------------- */
   509→      /*  EmailJS integration                                       */
   510→      /*  If credentials are placeholders, we simulate a success    */
   511→      /*  so the UI can be previewed. Replace the constants above    */
   512→      /*  with real EmailJS values to enable actual sending.         */
   513→      /* ---------------------------------------------------------- */
   514→      const isPlaceholder =
   515→        EMAILJS_PUBLIC_KEY.includes("YOUR_") ||
   516→        EMAILJS_SERVICE_ID.includes("YOUR_") ||
   517→        EMAILJS_TEMPLATE_ID.includes("YOUR_");
   518→
   519→      if (!isPlaceholder) {
   520→        const w = window as unknown as Record<string, unknown>;
   521→        const emailjs = w.emailjs as { init: (key: string) => void; send: (service: string, template: string, params: Record<string, string>) => Promise<unknown> };
   522→        emailjs.init(EMAILJS_PUBLIC_KEY);
   523→        await emailjs.send(
   524→          EMAILJS_SERVICE_ID,
   525→          EMAILJS_TEMPLATE_ID,
   526→          {
   527→            from_name: form.name,
   528→            from_phone: form.phone,
   529→            from_email: form.email,
   530→            service: form.service,
   531→            preferred_date: form.preferredDate,
   532→            message: form.message,
   533→          }
   534→        );
   535→      }
   536→
   537→      setSubmitted(true);
   538→      toast({
   539→        title: isPlaceholder
   540→          ? "Demo mode — email not sent"
   541→          : "Appointment requested!",
   542→        description: isPlaceholder
   543→          ? "Connect EmailJS to enable real email notifications."
   544→          : `We'll confirm your ${form.service} appointment shortly.`,
   545→      });
   546→    } catch {
   547→      toast({
   548→        title: "Something went wrong",
   549→        description:
   550→          "Please try again or call us directly.",
   551→        variant: "destructive",
   552→      });
   553→    } finally {
   554→      setSubmitting(false);
   555→    }
   556→  };
   557→
   558→  const updateField = (field: keyof FormData, value: string) => {
   559→    setForm((prev) => ({ ...prev, [field]: value }));
   560→    if (errors[field]) setErrors((prev) => ({ ...prev, [field]: undefined }));
   561→  };
   562→
   563→  return (
   564→    <section
   565→      id="appointment"
   566→      ref={sectionRef}
   567→      className="bg-white py-20 sm:py-28"
   568→    >
   569→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   570→        <motion.div
   571→          initial={{ opacity: 0, y: 20 }}
   572→          whileInView={{ opacity: 1, y: 0 }}
   573→          viewport={{ once: true, margin: "-80px" }}
   574→          transition={{ duration: 0.5 }}
   575→          className="mb-14 text-center"
   576→        >
   577→          <span className="mb-2 inline-block text-sm font-semibold tracking-wider text-primary uppercase">
   578→            Get Started
   579→          </span>
   580→          <h2 className="mb-4 text-3xl font-bold tracking-tight text-slate-900 sm:text-4xl">
   581→            Request an Appointment
   582→          </h2>
   583→          <p className="mx-auto max-w-2xl text-base text-slate-500">
