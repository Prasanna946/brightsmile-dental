   451→
   452→interface FormData {
   453→  name: string;
   454→  phone: string;
   455→  email: string;
   456→  service: string;
   457→  preferredDate: string;
   458→  message: string;
   459→}
   460→
   461→function AppointmentSection({
   462→  sectionRef,
   463→}: {
   464→  sectionRef: React.RefObject<HTMLDivElement | null>;
   465→}) {
   466→  const { toast } = useToast();
   467→  const [form, setForm] = useState<FormData>({
   468→    name: "",
   469→    phone: "",
   470→    email: "",
   471→    service: "",
   472→    preferredDate: "",
   473→    message: "",
   474→  });
   475→  const [errors, setErrors] = useState<Partial<FormData>>({});
   476→  const [submitting, setSubmitting] = useState(false);
   477→  const [submitted, setSubmitted] = useState(false);
   478→
   479→  const validate = (): boolean => {
   480→    const e: Partial<FormData> = {};
   481→    if (!form.name.trim()) e.name = "Please enter your name.";
   482→    if (!form.phone.trim()) e.phone = "Please enter your phone number.";
   483→    else if (!/^\d{10}$/.test(form.phone.trim())) e.phone = "Enter a valid 10-digit Indian number.";
   484→    if (!form.email.trim()) {
   485→      e.email = "Please enter your email.";
   486→    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
   487→      e.email = "Please enter a valid email.";
   488→    }
   489→    if (!form.service) e.service = "Please select a service.";
   490→    setErrors(e);
   491→    return Object.keys(e).length === 0;
   492→  };
   493→
   494→  const handleSubmit = async (e: FormEvent) => {
   495→    e.preventDefault();
   496→    if (!validate()) return;
   497→
   498→    setSubmitting(true);
   499→
   500→    try {
   501→      const res = await fetch("/api/appointment", {
   502→        method: "POST",
   503→        headers: { "Content-Type": "application/json" },
   504→        body: JSON.stringify({
   505→          name: form.name,
   506→          phone: `+91 ${form.phone}`,
   507→          email: form.email,
   508→          service: form.service,
   509→          preferredDate: form.preferredDate || undefined,
   510→          message: form.message || undefined,
   511→        }),
   512→      });
   513→
   514→      const data = await res.json();
   515→
   516→      if (!res.ok) {
   517→        toast({
   518→          title: "Submission failed",
   519→          description: data.error || "Please check the form and try again.",
   520→          variant: "destructive",
   521→        });
   522→        return;
   523→      }
   524→
   525→      setSubmitted(true);
   526→
   527→      if (data.emailSent) {
   528→        toast({
   529→          title: "Appointment requested!",
   530→          description: `We'll confirm your ${form.service} appointment shortly.`,
   531→        });
   532→      } else if (data.emailConfigured) {
   533→        toast({
   534→          title: "Appointment saved!",
   535→          description: "Email delivery failed, but we have your details. We'll call you to confirm.",
   536→          variant: "destructive",
   537→        });
   538→      } else {
   539→        toast({
   540→          title: "Appointment saved!",
   541→          description: "We'll contact you shortly to confirm your " + form.service + " appointment.",
   542→        });
   543→      }
   544→    } catch {
   545→      toast({
   546→        title: "Something went wrong",
   547→        description: "Please try again or call us directly.",
   548→        variant: "destructive",
   549→      });
   550→    } finally {
   551→      setSubmitting(false);
   552→    }
   553→  };
   554→
   555→  const updateField = (field: keyof FormData, value: string) => {
   556→    setForm((prev) => ({ ...prev, [field]: value }));
   557→    if (errors[field]) setErrors((prev) => ({ ...prev, [field]: undefined }));
   558→  };
   559→
   560→  return (
   561→    <section
   562→      id="appointment"
   563→      ref={sectionRef}
   564→      className="bg-white py-20 sm:py-28"
   565→    >
   566→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   567→        <motion.div
   568→          initial={{ opacity: 0, y: 20 }}
   569→          whileInView={{ opacity: 1, y: 0 }}
   570→          viewport={{ once: true, margin: "-80px" }}
   571→          transition={{ duration: 0.5 }}
   572→          className="mb-14 text-center"
   573→        >
   574→          <span className="mb-2 inline-block text-sm font-semibold tracking-wider text-primary uppercase">
   575→            Get Started
   576→          </span>
   577→          <h2 className="mb-4 text-3xl font-bold tracking-tight text-slate-900 sm:text-4xl">
   578→            Request an Appointment
   579→          </h2>
   580→          <p className="mx-auto max-w-2xl text-base text-slate-500">
   581→            Fill out the form below and we&apos;ll get back to you within one
   582→            business day to confirm your visit.
   583→          </p>
   584→        </motion.div>
   585→
   586→        <motion.div
   587→          initial={{ opacity: 0, y: 24 }}
   588→          whileInView={{ opacity: 1, y: 0 }}
   589→          viewport={{ once: true, margin: "-40px" }}
   590→          transition={{ duration: 0.5, delay: 0.1 }}
